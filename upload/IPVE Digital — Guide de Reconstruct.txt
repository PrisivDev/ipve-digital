6.2 src/lib/api-key.ts — Validation Clé API REST
typescript

import { prisma } from './prisma'
import bcrypt from 'bcryptjs'

export async function validateApiKey(apiKey: string): Promise<{ valid: boolean; name?: string; error?: string }> {
  if (!apiKey) {
    return { valid: false, error: 'API key is required' }
  }

  // Extraire le préfixe (8 premiers caractères) pour lookup rapide
  const prefix = apiKey.substring(0, 8)

  const storedKey = await prisma.apiKey.findFirst({
    where: {
      keyPrefix: prefix,
      isActive: true,
    },
  })

  if (!storedKey) {
    return { valid: false, error: 'Invalid API key' }
  }

  if (storedKey.expiresAt && storedKey.expiresAt < new Date()) {
    return { valid: false, error: 'API key has expired' }
  }

  // Vérifier le hash complet
  const isValid = await bcrypt.compare(apiKey, storedKey.keyHash)

  if (!isValid) {
    return { valid: false, error: 'Invalid API key' }
  }

  // Mettre à jour lastUsedAt
  await prisma.apiKey.update({
    where: { id: storedKey.id },
    data: { lastUsedAt: new Date() },
  })

  return { valid: true, name: storedKey.name }
}

export function extractApiKey(request: Request): string | null {
  // Priorité 1: Header Authorization: Bearer <key>
  const authHeader = request.headers.get('authorization')
  if (authHeader?.startsWith('Bearer ')) {
    const key = authHeader.substring(7)
    if (key.startsWith('ipve_')) return key
  }

  // Priorité 2: Header X-Api-Key
  const apiKeyHeader = request.headers.get('x-api-key')
  if (apiKeyHeader?.startsWith('ipve_')) return apiKeyHeader

  // Priorité 3: Query param ?api_key=<key>
  const url = new URL(request.url)
  const queryKey = url.searchParams.get('api_key')
  if (queryKey?.startsWith('ipve_')) return queryKey

  return null
}

// Flag pour éviter les boucles de refresh
let isRefreshing = false
let refreshPromise: Promise<string | null> | null = null

async function getRefreshedToken(): Promise<string | null> {
  if (isRefreshing && refreshPromise) return refreshPromise

  isRefreshing = true
  refreshPromise = (async () => {
    try {
      const res = await fetch('/api/auth/session', { method: 'GET' })
      if (!res.ok) return null
      const session = await res.json()
      return session?.accessToken || null
    } catch {
      return null
    } finally {
      isRefreshing = false
      refreshPromise = null
    }
  })()

  return refreshPromise
}

export async function apiFetch<T = unknown>(url: string, options: FetchOptions = {}): Promise<T> {
  const { skipAuth, ...fetchOptions } = options

  // Récupérer le token depuis la session NextAuth
  const sessionRes = await fetch('/api/auth/session')
  const session = await sessionRes.json()
  let token = session?.accessToken

  const headers = new Headers(fetchOptions.headers)
  headers.set('Content-Type', 'application/json')

  if (token && !skipAuth) {
    headers.set('Authorization', `Bearer ${token}`)
  }

  let response = await fetch(url, { ...fetchOptions, headers })

  // Si 401, tenter un refresh du token
  if (response.status === 401 && !skipAuth) {
    const newToken = await getRefreshedToken()
    if (newToken) {
      headers.set('Authorization', `Bearer ${newToken}`)
      response = await fetch(url, { ...fetchOptions, headers })

      if (response.status === 401) {
        // Second 401 → force logout
        await forceLogout()
        throw new Error('Session expirée. Veuillez vous reconnecter.')
      }
    } else {
      await forceLogout()
      throw new Error('Session expirée. Veuillez vous reconnecter.')
    }
  }

  if (!response.ok) {
    const error = await response.json().catch(() => ({ error: 'Erreur réseau' }))
    throw new Error(error.error || `Erreur ${response.status}`)
  }

  return response.json()
}
6.4 src/lib/verify-admin.ts — Middleware Vérification Admin
typescript

import { getServerSession } from 'next-auth'
import { authOptions } from './auth-options'
import { NextResponse } from 'next/server'

export async function verifySettingsAdmin() {
  const session = await getServerSession(authOptions)

  if (!session?.user) {
    return { error: 'Non authentifié', status: 401, response: NextResponse.json({ error: 'Non authentifié' }, { status: 401 }) }
  }

  if (session.user.role !== 'admin') {
    return { error: 'Accès non autorisé', status: 403, response: NextResponse.json({ error: 'Accès non autorisé' }, { status: 403 }) }
  }

  return { session, response: null }
}

export async function GET(request: NextRequest) {
  // Validation clé API
  const apiKey = extractApiKey(request)
  const authResult = await validateApiKey(apiKey || '')
  if (!authResult.valid) {
    return json({ error: authResult.error }, 401)
  }

  try {
    const { searchParams } = new URL(request.url)
    const page = parseInt(searchParams.get('page') || '1')
    const limit = parseInt(searchParams.get('limit') || '20')
    const status = searchParams.get('status')
    const source = searchParams.get('source')

    const where: any = {}
    if (status) where.status = status
    if (source) where.source = source

    const [admissions, total] = await Promise.all([
      prisma.admission.findMany({
        where,
        include: {
          filiere: { select: { id: true, name: true, code: true } },
          niveau: { select: { id: true, name: true } },
          prospect: { select: { id: true, status: true } },
        },
        orderBy: { createdAt: 'desc' },
        skip: (page - 1) * limit,
        take: limit,
      }),
      prisma.admission.count({ where }),
    ])

    return json({
      success: true,
      data: admissions,
      pagination: {
        page,
        limit,
        total,
        pages: Math.ceil(total / limit),
      },
    })
  } catch (error: any) {
    return json({ error: 'Erreur interne du serveur' }, 500)
  }
}

11. BUGS CONNUS ET CORRECTIONS APPLIQUÉES
11.1 instanceof NextResponse toujours false (CORRIGÉ)
Problème : Dans les API routes, response instanceof NextResponse retournait toujours false car l'objet est sérialisé/cross-realm.

Correction : Remplacé instanceof NextResponse par instanceof Response dans 26 fichiers API routes.

typescript

// ❌ AVANT (ne fonctionne pas)
if (result.response instanceof NextResponse) {
  return result.response
}

// ✅ APRÈS (fonctionne)
if (result.response instanceof Response) {
  return result.response
}
11.2 Boucle infinie de refresh token 401 (CORRIGÉ)
Problème : Lorsque le token expire, api-fetch.ts tente un refresh, mais si le refresh échoue, cela créait une boucle infinie de tentatives.

Correction : Ajout d'un flag isRefreshing et d'une limite de 2 tentatives max avant forceLogout().

11.3 Désynchronisation Prisma ↔ Supabase (CORRIGÉ)
Problème : Après modification du schéma, Prisma et Supabase n'étaient plus synchronisés.

Correction : bunx prisma db push --accept-data-loss (en production, sauvegarder d'abord !)

11.4 Endpoints 500 au premier appel (CORRIGÉ)

Cause : Délai de compilation JIT de Next.js en production.

Solution : Pas de bug réel, comportement normal. Pré-warm les endpoints critiques.

      name: 'Administrateur IPVE',
      password: hashedPassword,
      role: 'admin',

  })

  const keyHash = await bcrypt.hash(apiKey, 12)
